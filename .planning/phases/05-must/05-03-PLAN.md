        """D-10 level 1 — 직전 level 0 rows. 키 (user_key, snapshot_id, variant), TTL CACHE_TTL_S. dict+Lock(데코레이터 금지)."""
        def put(self, user_key, snapshot_id, variant, rows, model_version):
            with self._lock:
                self._d[(user_key, snapshot_id, variant)] = (self._now() + self._ttl, tuple(rows), model_version)
        def get(self, user_key, snapshot_id, variant):
            key = (user_key, snapshot_id, variant)
            with self._lock:
                hit = self._d.get(key)
                if hit is None: return None
                exp, rows, version = hit
