    # walk.py <base>: 콘솔 에러·pageerror 수집, 완주 10단계, variant 전환 4회 → 카드 제목 목록 비교
    import sys, json; from playwright.sync_api import sync_playwright as sp
    base = sys.argv[1].rstrip("/"); errs = []
    def titles(pg): return [t.inner_text().split("\n")[0] for t in pg.query_selector_all(".tile")][:10]
    with sp() as p:
        b = p.chromium.launch(); pg = b.new_page(viewport={"width": 1280, "height": 900})
        pg.on("console", lambda m: errs.append(m.text) if m.type == "error" else None)
        pg.on("pageerror", lambda e: errs.append(str(e)))
        pg.goto(f"{base}/?source=api#/"); pg.wait_for_load_state("networkidle")
        for route in ["#/onboarding", "#/home", "#/book/1", "#/reader/1", "#/home", "#/library", "#/refresh", "#/home", "#/dashboard"]:
            pg.goto(f"{base}/?source=api{route}"); pg.wait_for_load_state("networkidle"); pg.wait_for_timeout(600)
        # 실제 화면 흐름(온보딩 7단계 클릭 → 홈) 은 06-06 체크리스트의 data-act 로 보강: 최소 온보딩 완료 후 #/home 카드 ≥ 5
        pg.goto(f"{base}/?source=api#/home"); pg.wait_for_selector(".tile", timeout=10000)
        seen = {}
        for v in ["pop", "cf", "hybrid", "hybrid_div"]:
            r = pg.query_selector(f'input[name="model"][value="{v}"]')
            if r is None: errs.append(f"no radio for {v}"); continue
            r.check(); pg.wait_for_load_state("networkidle"); pg.wait_for_timeout(800); seen[v] = titles(pg)
        distinct = len({tuple(t) for t in seen.values()})
        print(json.dumps({"console_errors": errs, "variants_seen": list(seen), "distinct_title_lists": distinct, "titles": seen}, ensure_ascii=False, indent=1))
